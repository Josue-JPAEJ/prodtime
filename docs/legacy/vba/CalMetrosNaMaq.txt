Private Sub ProducaoEstimada()
    'On Error GoTo Erro
    Dim lProdEstimada As Long, lSaldo As Long, m_h As Double
    Dim IntervaloDeDias As Integer
    Dim ContadorSabados As Integer, ContadorDomingos As Integer
    Dim DateIni As Date, ContadorFeriados As Integer, dFeriado As String
    Dim i As Integer

    LimparVisor

    If dDesperd = Empty Or HorasUteis = Empty Or DataEntrega = Empty Or DataInicial = Empty Or CM_MIN = Empty Then Exit Sub
    IntervaloDeDias = (DataEntrega + 1) - (DataInicial)
    ContadorDomingos = 0
    ContadorSabados = 0
    ContadorFeriados = 0
    For DateIni = DataInicial To DataEntrega
        If Weekday(DateIni) = 1 Then ContadorDomingos = ContadorDomingos + 1
        If Weekday(DateIni) = 7 Then ContadorSabados = ContadorSabados + 1
        For i = 0 To Me.lbFeriados.ListCount - 1
            dFeriado = Left(Me.lbFeriados.list(i, 1), Len(Me.lbFeriados.list(i, 1)) - 4) & Year(DateIni)
            If DateIni = dFeriado Then ContadorFeriados = ContadorFeriados + 1
        Next
    Next
    If Feriado = False Then IntervaloDeDias = IntervaloDeDias - ContadorFeriados
    If Sabado = False Then IntervaloDeDias = IntervaloDeDias - ContadorSabados
    If Domingo = False Then IntervaloDeDias = IntervaloDeDias - ContadorDomingos
    If bDatInic = False Then IntervaloDeDias = IntervaloDeDias - 1
    If bDatEnt = False Then IntervaloDeDias = IntervaloDeDias - 1

    m_h = (CM_MIN * 60) / 100
    'pE=m_h*qntFitas*huteis*qntdias-desperdicio
    lProdEstimada = m_h * qntFita * HorasUteis * IntervaloDeDias

    'Visor Relatorio  #########################################
    Me.chbFeriados.caption = ContadorFeriados & " - " & Me.chbFeriados.tag
    Me.chbSab.caption = ContadorSabados & " - " & Me.chbSab.tag
    Me.chbDom.caption = ContadorDomingos & " - " & Me.chbDom.tag
    Me.tbTotalDias = IntervaloDeDias
    Me.tbTotalHoras = HorasUteis * IntervaloDeDias
    Me.tbTotalDesperdicio = (lProdEstimada * (dDesperd / 100))
    '_________________________________________________________

    lProdEstimada = lProdEstimada - (lProdEstimada * (dDesperd / 100))
    Me.tbProdEstimada = lProdEstimada
    If metrosCalcular = Empty Then Exit Sub
    Me.tbSaldo = lProdEstimada - metrosCalcular
    Me.caption = Me.tag & "   -   Login: " & Login
    Exit Sub
Erro:
    Call ContemErros(Me.Name, "ProducaoEstimada")
End Sub

Private Sub LimparVisor()
    'On Error GoTo Erro

    Me.chbSab.caption = Me.chbSab.tag
    Me.chbDom.caption = Me.chbDom.tag

    Exit Sub
Erro:
    Call ContemErros(Me.Name, "LimparVisor")
End Sub

Private Sub TotalMetros()
    Dim tBruto As Double, tTara As Double, tLiq As Double
    'On Error GoTo Erro

    'Total PesoLiquido
    If tara = Empty Or pBruto = Empty Then Exit Sub
    pLiq = pBruto - tara
    Me.tbPesoLiq = pLiq
    Me.tbPesoLiq = Left(Me.tbPesoLiq, InStr(1, Me.tbPesoLiq, ",", vbBinaryCompare) + 3)
    If InStr(1, Me.tbPesoLiq, ",", vbBinaryCompare) = 0 Then Me.tbPesoLiq = Me.tbPesoLiq & ",000"

    'Total TotalMetros
    If qntCaixas = Empty Or pFita = Empty Then Exit Sub
    tBruto = pBruto * qntCaixas
    tTara = tara * qntCaixas
    tLiq = tBruto - tTara
    tMetros = (tLiq * 1000) / pFita
    Me.tbTotal = tMetros & "m"

    Exit Sub
Erro:
    Call ContemErros(Me.Name, "TotalMetros")
End Sub

Private Sub itbDataEntrega_Exit()

    If Me.tbDataEntrega = "" Then
        DataEntrega = Empty

        Exit Sub

        Me.tbDataEntrega = ""
        DataEntrega = Empty
        Me.tbTotalDias = ""
        Exit Sub

        DataEntrega = Me.tbDataEntrega
        If DataInicial >= DataEntrega And Me.tbDataInicial <> "" Then Me.tbDataEntrega = ""
    End If
    ProducaoEstimada

    Exit Sub

Private Sub itbPesoFita_Exit()
    'On Error GoTo Erro

    If Me.tbPesoFita = "" Then
        pFita = Empty
        Exit Sub
    End If
    If Right(Me.tbPesoFita.Text, 1) = "," Then
        Me.tbPesoFita = Me.tbPesoFita & "00"
    End If
    If Me.tbPesoFita <> "" Then
        If InStr(1, Me.tbPesoFita, ",", vbTextCompare) = 0 Then
            Me.tbPesoFita = Me.tbPesoFita & ",00"
        End If
        pFita = Replace(Me.tbPesoFita, ".", ",")
        Me.tbPesoFita = Me.tbPesoFita & "g"
        TotalMetros
    End If

    Exit Sub
Erro:
    Call ContemErros(Me.Name, "itbPesoFita_Exit")

Private Sub ListFeriados()
    Dim i As Integer
    'On Error GoTo Erro

    With Me.lbFeriados
        .Clear
        i = 0
        Do Until Rel.Cells(i + 2, 1) = ""
            .AddItem
            .list(i, 0) = Rel.Cells(i + 2, 1)
            .list(i, 1) = Rel.Cells(i + 2, 2) & "/" & Year(VBA.DateValue(Now()))
            .list(i, 2) = Rel.Cells(i + 2, 3)
            i = i + 1
        Loop
    End With

    Exit Sub
Erro:
    Call ContemErros(Me.Name, "ListFeriados")
End Sub
